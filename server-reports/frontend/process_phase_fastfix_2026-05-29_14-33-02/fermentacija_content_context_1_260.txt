    17	function dcpf_is_page(): bool {
    18	    $path = trim(parse_url($_SERVER['REQUEST_URI'] ?? '', PHP_URL_PATH), '/');
    19	    return $path === 'proces-izrade/fermentacija';
    20	}

    22	function dcpf_page_url(string $path): string {
    23	    $page = get_page_by_path($path);
    24	    if ($page && $page->post_status === 'publish') {
    25	        return get_permalink($page);
    26	    }
    27	    return home_url('/' . trim($path, '/') . '/');
    28	}

   213	function dcpf_append_to_page($content) {
   214	    static $added = false;
   215	
   216	    if ($added || !dcpf_enabled()) {
   217	        return $content;
   218	    }
   219	
   220	    if (!dcpf_is_page() || !in_the_loop() || !is_main_query()) {
   221	        return $content;
   222	    }
   223	
   224	    if (strpos($content, 'dcpf-wrap') !== false) {
   225	        return $content;
   226	    }
   227	
   228	    $added = true;
   229	    return $content . do_shortcode('[drycured_process_fermentacija]');
   230	}
   231	add_filter('the_content', 'dcpf_append_to_page', 35);
